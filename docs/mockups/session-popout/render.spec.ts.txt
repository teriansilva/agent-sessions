import { expect, test, type Page } from "@playwright/test";
import { STRUCTURED_ID } from "./apiLookFixture";
const OUT = "/tmp/popout-shots";
// Reuse the spec's setup by importing it would execute its tests; copy minimal bits instead.
import * as spec from "./sessionPopout";

for (const theme of ["dark", "light"] as const) {
  test(`shots ${theme}`, async ({ page }, ti) => {
    const proj = ti.project.name;
    await page.context().addInitScript((t) => localStorage.setItem("tr-theme", t), theme);
    await spec.setup(page);
    await page.goto(`/s/codex-api/${STRUCTURED_ID}`);
    await expect(page.getByTestId("structured-pane")).toBeVisible();
    await expect(page.getByTestId("md-code").first()).toBeVisible();
    const menu = await spec.openPaneMenu(page);
    await expect(menu.getByRole("menuitem", { name: "Pop out session into its own window" })).toBeVisible();
    await page.waitForTimeout(600);
    await page.screenshot({ path: `${OUT}/${proj}-${theme}-1-menu.png` });
    const [popup] = await Promise.all([
      page.context().waitForEvent("page"),
      menu.getByRole("menuitem", { name: "Pop out session into its own window" }).click(),
    ]);
    if (proj === "desktop") await popup.setViewportSize({ width: 1100, height: 760 });
    await expect(popup.getByTestId("md-code").first()).toBeVisible();
    await popup.waitForTimeout(800);
    await popup.screenshot({ path: `${OUT}/${proj}-${theme}-2-popout-api.png` });
    const p2 = await page.context().newPage();
    if (proj === "desktop") await p2.setViewportSize({ width: 1100, height: 760 });
    await p2.goto(`/pop/claude/${spec.PTY_ID}`);
    await expect(p2.locator(".xterm")).toBeVisible();
    await p2.waitForTimeout(1200);
    await p2.screenshot({ path: `${OUT}/${proj}-${theme}-3-popout-term.png` });
  });
}
